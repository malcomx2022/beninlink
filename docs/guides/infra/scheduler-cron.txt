# --- File des envois (D13) ---------------------------------------------------
# Repli SANS superviseur seulement — c'est l'un OU l'autre, jamais les deux.
# Le worker démarre chaque minute, vide la file et s'arrête : un envoi peut

# * * * * * cd /var/www/beninlink/web && php artisan queue:work --stop-when-empty --tries=3 >> /dev/null 2>&1

# --- Supervision -------------------------------------------------------------
# `beninlink-surveiller` n'écrit et n'alerte QUE sur échec : une alerte qui
# arrive tous les jours cesse d'être lue en une semaine.
#
# Ces trois commandes sont les SEULES dont la sortie 1 signale un incident.
# Les six autres réservent leur code 1 aux erreurs d'appel, et deux constats
# renvoient SUCCESS même quand ils trouvent des écarts : les brancher ici
# produirait un vert permanent. Voir supervision/README.md.

# Worker arrêté : l'application répond, et plus aucun SMS ne part.
*/10 * * * * /usr/local/bin/beninlink-surveiller beninlink:file-attente >> /var/log/beninlink-supervision.log 2>&1

# Une société ne peut plus facturer : la création de colis est refusée.
30 6 * * * /usr/local/bin/beninlink-surveiller beninlink:tarification-prete >> /var/log/beninlink-supervision.log 2>&1
